cmake_minimum_required(VERSION 3.16)
project(dzo_perspective LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
add_executable(dzo_perspective main.cpp)
target_include_directories(dzo_perspective PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(dzo_perspective PRIVATE ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dzo_perspective PRIVATE -Wall -Wextra -Wpedantic)
endif()
